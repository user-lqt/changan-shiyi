func _ready() -> void:
	face = load("res://assets/fonts/ui.ttc") as Font
	if ResourceLoader.exists("res://assets/images/workshop.png"):
		art = load("res://assets/images/workshop.png") as Texture2D
	title_art = _try_load("res://assets/images/title.jpg")
	paper_art = _try_load("res://assets/images/paper.jpg")
	mirror_art = _try_load("res://assets/images/mirror.png")
	shard_art = _try_load("res://assets/images/shard.png")
	var args := OS.get_cmdline_user_args()
	test_mode = args.has("--self-test") or args.has("--ui-test")
	transient_save = test_mode or args.has("--fresh")
	for arg in args:
		if arg.begins_with("--capture="):
			capture_path = arg.trim_prefix("--capture=")
		if arg.begins_with("--stage="):
			capture_stage = arg.trim_prefix("--stage=")
	if not capture_path.is_empty():
		transient_save = true
	var rng := RandomNumberGenerator.new()
	rng.seed = 9173
	for i in range(42):
		motes.append(Vector3(rng.randf_range(0, 1074), rng.randf_range(0, 604), rng.randf_range(0.4, 1.6)))
	for i in range(140):
		rain.append(Vector3(rng.randf(), rng.randf(), rng.randf_range(0.6, 1.4)))
	_prepare_audio()
	if not transient_save:
		_load_progress()
	angle_vis = float(angle_step)
	lid_t = 1.0 if cabinet_open else 0.0
	if args.has("--ui-test"):
		_run_ui_test.call_deferred()
	elif test_mode:
		_run_self_test.call_deferred()
	elif not capture_path.is_empty():
		_capture.call_deferred()
	queue_redraw()


func _try_load(path: String) -> Texture2D:
	return load(path) as Texture2D if ResourceLoader.exists(path) else null


func _prepare_audio() -> void:
	for track in ["ambient", "rain", "click", "hover", "pickup", "place", "rotate", "dial", "note", "error", "reveal", "unlock", "win", "intro", "revealed", "complete"]:
		var file := "res://assets/audio/%s.wav" % track
		if ResourceLoader.exists(file):
			tracks[track] = load(file)
	ambient = AudioStreamPlayer.new()
	ambient.volume_db = -13
	add_child(ambient)
	if tracks.has("ambient"):
		ambient.stream = tracks["ambient"]
		ambient.finished.connect(func() -> void:
			if sound_on:
				ambient.play())
		ambient.play()
	rain_player = AudioStreamPlayer.new()
	rain_player.volume_db = -20
	add_child(rain_player)
	if tracks.has("rain"):
		rain_player.stream = tracks["rain"]
		rain_player.finished.connect(func() -> void:
			if sound_on:
				rain_player.play())
		rain_player.play()
	speech = AudioStreamPlayer.new()
	speech.volume_db = -3
	add_child(speech)
	for i in range(6):
		var player := AudioStreamPlayer.new()
		player.volume_db = -6
		add_child(player)
		sound_players.append(player)


func _sfx(name: String) -> void:
	if not sound_on or not tracks.has(name) or test_mode:
		return
	for player in sound_players:
		if not player.playing:
			player.stream = tracks[name]
			player.play()
			return


func _narrate(name: String) -> void:
	if sound_on and tracks.has(name) and not test_mode:
		speech.stream = tracks[name]
		speech.play()


func _process(delta: float) -> void:
	age += delta
	if screen == "game" and not completed:
		play_time += delta
	toast_time = maxf(0.0, toast_time - delta)
	beam_flash = maxf(0.0, beam_flash - delta)
	journal_pulse = maxf(0.0, journal_pulse - delta)
	fade = maxf(0.0, fade - delta * 1.6)
	modal_t = minf(1.0, modal_t + delta * 5.0)
	shake = maxf(0.0, shake - delta * 2.5)
	lid_t = move_toward(lid_t, 1.0 if cabinet_open else 0.0, delta * 1.4)
	angle_vis = lerpf(angle_vis, float(angle_step), minf(1.0, delta * 12.0))
	for i in range(3):
		dial_flip[i] = maxf(0.0, float(dial_flip[i]) - delta * 4.0)
	for i in range(ripples.size() - 1, -1, -1):
		ripples[i].z += delta
		if ripples[i].z > 0.6:
			ripples.remove_at(i)
	queue_redraw()


func _set_modal(name: String) -> void:
	if name != modal:
		modal_t = 0.0 if not name.is_empty() else 1.0
	modal = name

func _cover_crop(size: Vector2, target: Vector2) -> Rect2:
	var scale_factor := maxf(target.x / size.x, target.y / size.y)
	var use := target / scale_factor
	return Rect2((size - use) * 0.5, use)


func _draw_lamp() -> void:
	for r in range(9, 0, -1):
		draw_circle(LAMP, r * 7, Color(1, 0.66, 0.24, 0.006))
	draw_circle(LAMP, 7 + sin(age * 5) * 0.8, Color("ffe7a3"))
	draw_circle(LAMP, 3, Color("fffbde"))


func _draw_stand() -> void:
	if mirror_placed:
		var incoming := (MIRROR - LAMP).normalized()
		var normal := (incoming - _outgoing(angle_vis)).normalized()
		var tangent := Vector2(-normal.y, normal.x)
		draw_line(MIRROR - tangent * 34, MIRROR + tangent * 34, Color("5b4b35"), 12, true)
		draw_line(MIRROR - tangent * 31, MIRROR + tangent * 31, GOLD, 7, true)
		draw_line(MIRROR - tangent * 27, MIRROR + tangent * 27, Color("cae0cc"), 3, true)
	else:
		draw_arc(MIRROR, 29, 0, TAU, 32, Color(0.9, 0.78, 0.5, 0.35), 2, true)
		draw_line(MIRROR + Vector2(-16, 0), MIRROR + Vector2(16, 0), Color(0.9, 0.78, 0.5, 0.25), 1)

func _draw_casket() -> void:
	var p := Vector2(925, 481)
	if lid_t > 0.0:
		var e := ease(lid_t, -2.0)
		# Dark interior first, then the lid lifts and slides back.
		_box(Rect2(p + Vector2(1, 5), Vector2(153, 37)), Color("12211d"), 3, Color("837348"))
		if not completed and e > 0.4:
			var glow := (e - 0.4) / 0.6
			for r in range(6, 0, -1):
				draw_circle(p + Vector2(76, 18), 18 + r * 7, Color(0.72, 1.0, 0.81, 0.012 * glow + 0.006 * sin(age * 2.0) * glow))
			_shard(p + Vector2(76, 18 - 6 * glow + sin(age * 1.6) * 1.5), 0.42)
		_box(Rect2(p + Vector2(1, 5 - 20 * e), Vector2(153, 18 + (1.0 - e) * 19)), Color("58442c"), 3, Color("a08a56"))
	else:
		for i in range(3):
			var flip := float(dial_flip[i])
			var h := 28.0 * (1.0 - flip * 0.6)
			_box(Rect2(p + Vector2(39 + i * 23, 24 - h * 0.5), Vector2(20, h)), Color("9e8858"), 2, Color("c6ad73"))
			if flip < 0.5:
				_glyph(p + Vector2(49 + i * 23, 24), int(dials[i]), 7, Color("3d3929"))


func _outgoing(step: float = -1.0) -> Vector2:
	var s := float(angle_step) if step < 0.0 else step
	return Vector2.from_angle(deg_to_rad(-110.0 + s * 15.0))


func _ray_end(start: Vector2, direction: Vector2) -> Vector2:
	var distance := 2000.0
	if direction.x > 0.001:
		distance = minf(distance, (ROOM.end.x - 3 - start.x) / direction.x)
	elif direction.x < -0.001:
		distance = minf(distance, (ROOM.position.x + 3 - start.x) / direction.x)
	if direction.y > 0.001:
		distance = minf(distance, (ROOM.end.y - 3 - start.y) / direction.y)
	elif direction.y < -0.001:
		distance = minf(distance, (ROOM.position.y + 3 - start.y) / direction.y)
	return start + direction * distance


func _draw_light() -> void:
	var dir := _outgoing(angle_vis)
	var end := TARGET if revealed else _ray_end(MIRROR, dir)
	var shimmer := 0.85 + 0.15 * sin(age * 6.0)
	for w in [22.0, 12.0, 5.0, 1.5]:
		var a := (0.027 if w > 5 else 0.36) * shimmer
		draw_line(LAMP, MIRROR, Color(1, 0.8, 0.42, a), w, true)
		draw_line(MIRROR, end, Color(1, 0.85, 0.52, a), w, true)
	# Motes of light travelling along both beam segments.
	for i in range(10):
		var k := fmod(age * 0.35 + i * 0.1, 1.0)
		var seg_a := LAMP.lerp(MIRROR, k)
		var seg_b := MIRROR.lerp(end, k)
		draw_circle(seg_a, 1.6, Color(1, 0.95, 0.75, 0.5 * sin(k * PI)))
		draw_circle(seg_b, 1.8, Color(1, 0.95, 0.75, 0.55 * sin(k * PI)))
	# Light pool where the beam lands.
	for r in range(5, 0, -1):
		draw_circle(end, r * 6, Color(1, 0.85, 0.5, 0.03))
	draw_circle(end, 4, Color(1, 0.95, 0.78, 0.7))
	# Gentle cue that the beam is getting close to the plaque.
	if not revealed:
		var along := (TARGET - MIRROR).dot(dir)
		var miss := (MIRROR + dir * maxf(0, along)).distance_to(TARGET)
		if along > 0 and miss < 120:
			draw_arc(TARGET, 62, 0, TAU, 64, Color(1, 0.85, 0.5, (1.0 - miss / 120.0) * 0.45), 2.0, true)
	draw_circle(MIRROR, 6, Color("ffe8a9"))
	if beam_flash > 0:
		draw_circle(TARGET, 75 + (2 - beam_flash) * 65, Color(1, 0.87, 0.5, beam_flash * 0.05))

		return 2

func _corner_ornaments(r: Rect2) -> void:
	# Small Tang-style bracket corners framing every panel.
	var c := Color(0.84, 0.72, 0.48, 0.55)
	for corner in [r.position, Vector2(r.end.x, r.position.y), r.end, Vector2(r.position.x, r.end.y)]:
		var sx := 1.0 if corner.x < r.get_center().x else -1.0
		var sy := 1.0 if corner.y < r.get_center().y else -1.0
		var o: Vector2 = corner + Vector2(10 * sx, 10 * sy)
		draw_line(o, o + Vector2(26 * sx, 0), c, 1.5, true)
		draw_line(o, o + Vector2(0, 26 * sy), c, 1.5, true)
		draw_line(o + Vector2(5 * sx, 5 * sy), o + Vector2(14 * sx, 5 * sy), c, 1.0, true)
		draw_line(o + Vector2(5 * sx, 5 * sy), o + Vector2(5 * sx, 14 * sy), c, 1.0, true)

func _glyph(p: Vector2, symbol: int, size: float, color: Color) -> void:
	var points := PackedVector2Array()
	match symbol:
		0:
			points = PackedVector2Array([Vector2(-1, 0.7), Vector2(-1, -0.3), Vector2(-0.8, 0.7), Vector2(0, 0.7), Vector2(0, -0.9), Vector2(0, 0.7), Vector2(1, 0.7), Vector2(1, -0.3)])
		1:
			for i in range(24):
				var a := -PI * 0.53 + float(i) / 23.0 * PI * 1.45
				points.append(Vector2(cos(a), sin(a)))
			points.append(Vector2(0.05, -0.1))
			points.append(points[0])
		2:
			for j in range(3):
				var wave := PackedVector2Array()
				for i in range(18):
					wave.append(p + Vector2(-1.0 + i / 17.0 * 2, (j - 1) * 0.58 + sin(i / 17.0 * TAU) * 0.14) * size)
				draw_polyline(wave, color, maxf(1.2, size * 0.09), true)
			return
		3:
			draw_arc(p, size * 0.8, 0, TAU, 32, color, maxf(1.2, size * 0.1), true)
			for i in range(8):
				var dir := Vector2.from_angle(float(i) * TAU / 8)
				draw_line(p + dir * size, p + dir * size * 1.25, color, maxf(1.2, size * 0.08), true)
			return
		4:
			points = PackedVector2Array([Vector2(0, 1), Vector2(0, -1), Vector2(-.9, .3), Vector2(0, -.5), Vector2(.9, .3), Vector2(0, -.5)])
		5:
			for i in range(3):
				draw_arc(p + Vector2((i - 1) * size * .6, 0), size * .6, PI, TAU, 20, color, maxf(1.2, size * .1), true)
			draw_line(p + Vector2(-1.6, .3) * size, p + Vector2(1.6, .3) * size, color, maxf(1.2, size * .1), true)
			return
	for i in range(points.size()):
		points[i] = p + points[i] * size
	if points.size() > 1:
		draw_polyline(points, color, maxf(1.2, size * 0.09), true)

func _wrap(p: Vector2, text: String, width: float, size: int, color: Color, line_height: float) -> void:
	var line := ""
	var row := 0
	for character in text:
		if character == "\n" or face.get_string_size(line + character, HORIZONTAL_ALIGNMENT_LEFT, -1, size).x > width:
			_t(p + Vector2(0, row * line_height), line, size, color)
			row += 1
			line = "" if character == "\n" else character
		else:
			line += character
	_t(p + Vector2(0, row * line_height), line, size, color)

func _gui_input(event: InputEvent) -> void:
	if event is InputEventMouseMotion:
		var hover_id := ""
		for i in range(hit_regions.size() - 1, -1, -1):
			if (hit_regions[i]["rect"] as Rect2).has_point(event.position):
				hover_id = str(hit_regions[i]["id"])
				break
		if hover_id != old_hover and not hover_id.is_empty():
			_sfx("hover")
		old_hover = hover_id
		mouse_default_cursor_shape = Control.CURSOR_POINTING_HAND if not hover_id.is_empty() else Control.CURSOR_ARROW
	elif event is InputEventMouseButton and event.pressed:
		if event.button_index in [MOUSE_BUTTON_WHEEL_UP, MOUSE_BUTTON_WHEEL_DOWN]:
			if mirror_placed and not revealed and modal.is_empty() and screen == "game":
				_turn_mirror(1 if event.button_index == MOUSE_BUTTON_WHEEL_UP else -1)
			return
		if event.button_index == MOUSE_BUTTON_LEFT:
			for i in range(hit_regions.size() - 1, -1, -1):
				if (hit_regions[i]["rect"] as Rect2).has_point(event.position):
					ripples.append(Vector3(event.position.x, event.position.y, 0.0))
					_act(str(hit_regions[i]["id"]))
					accept_event()
					return
			# Clicking empty scenery still gives a light acknowledgement.
			if screen == "game" and modal.is_empty() and ROOM.has_point(event.position):
				ripples.append(Vector3(event.position.x, event.position.y, 0.3))


func _unhandled_key_input(event: InputEvent) -> void:
	if not event is InputEventKey or not event.pressed or event.echo:
		return
	if event.keycode == KEY_ESCAPE:
		_set_modal("" if not modal.is_empty() else "help")
		_sfx("click")
	elif screen == "title" and event.keycode in [KEY_ENTER, KEY_SPACE]:
		_act("start")
	elif screen == "game" and not modal.is_empty() and event.keycode in [KEY_ENTER, KEY_SPACE]:
		if modal == "seal":
			_act("check")
		else:
			_act("close")
	elif screen == "game" and modal == "seal":
		match event.keycode:
			KEY_1, KEY_2, KEY_3:
				_act("dial_%d" % (event.keycode - KEY_1))
	elif screen == "game" and modal.is_empty():
		match event.keycode:
			KEY_A, KEY_LEFT:
				_turn_mirror(-1)
			KEY_D, KEY_RIGHT:
				_turn_mirror(1)
			KEY_J:
				_act("journal")
			KEY_H:
				_act("hint")
	queue_redraw()


func _act(id: String) -> void:
	if not id.begins_with("dial_") and not id.begins_with("back_"):
		_sfx("click")
	match id:
		"sound":
			sound_on = not sound_on
			AudioServer.set_bus_mute(0, not sound_on)
			if sound_on:
				if tracks.has("ambient") and not ambient.playing:
					ambient.play()
				if tracks.has("rain") and not rain_player.playing:
					rain_player.play()
		"help":
			_set_modal("help")
		"start":
			screen = "game"
			fade = 1.0
			if not mirror_found:
				_narrate("intro")
		"new", "restart":
			_reset_progress()
			screen = "game"
			fade = 1.0
			_narrate("intro")
		"close":
			_set_modal("")
		"note":
			note_read = true
			_set_modal("note")
			_sfx("note")
			_save_progress()
		"pickup":
			if not mirror_found:
				mirror_found = true
				selected_item = true
				_set_modal("mirror")
				_notify("获得铜镜。点击空镜台，将它安放上去。")
				_sfx("pickup")
				_save_progress()
		"inventory":
			selected_item = not selected_item
		"stand":
			if not mirror_found:
				_notify("镜台空着。也许附近能找到合适的铜镜。")
			elif not selected_item:
				_notify("先点击行囊中的铜镜，再将它安放在这里。")
			else:
				_place_mirror()
		"turn_left":
			_turn_mirror(-1)
		"turn_right":
			_turn_mirror(1)
		"mural":
			if revealed:
				_set_modal("journal")
			else:
				_notify("石壁上有浅浅刻痕。也许需要更准确的一束光。")
		"chest":
			if cabinet_open:
				_finish()
			else:
				_set_modal("seal")
		"check":
			_check_seal()
		"journal":
			_set_modal("journal")
		"hint":
			_hint()
		_:
			if id.begins_with("dial_"):
				var i := int(id.trim_prefix("dial_"))
				dials[i] = (int(dials[i]) + 1) % 6
				dial_flip[i] = 1.0
				_sfx("dial")
				_save_progress()
			elif id.begins_with("back_"):
				var i := int(id.trim_prefix("back_"))
				dials[i] = posmod(int(dials[i]) - 1, 6)
				dial_flip[i] = 1.0
				_sfx("dial")
				_save_progress()
	queue_redraw()


func _place_mirror() -> void:
	if not mirror_found or mirror_placed:
		return
	mirror_placed = true
	selected_item = false
	angle_vis = float(angle_step)
	_sfx("place")
	_notify("铜镜已安放。用 A / D 或滚轮调整光线。")
	_save_progress()


func _turn_mirror(direction: int) -> void:
	if not mirror_placed or revealed:
		return
	angle_step = posmod(angle_step + direction, 11)
	_sfx("rotate")
	var ray := _outgoing()
	var along := (TARGET - MIRROR).dot(ray)
	var closest := MIRROR + ray * maxf(0, along)
	if along > 0 and closest.distance_to(TARGET) < 29:
		revealed = true
		beam_flash = 2.0
		journal_pulse = 3.0
		_notify("光照见了三枚纹样：山 → 月 → 水。线索已记入手札。", 6)
		_sfx("reveal")
		_narrate("revealed")
	_save_progress()


func _check_seal() -> void:
	if not revealed:
		_notify("先让石壁纹样显现，找到开启旧匣的依据。")
		_sfx("error")
		shake = 1.0
		return
	if dials == ANSWER:
		cabinet_open = true
		lid_t = 0.0
		_set_modal("")
		_notify("机关轻响，旧匣已开启。点击取出青瓷残片。", 5)
		_sfx("unlock")
		_save_progress()
	else:
		_notify("纹样尚未吻合。按石壁从左到右的顺序再试一次。")
		_sfx("error")
		shake = 1.0


func _finish() -> void:
	if not cabinet_open:
		return
	completed = true
	_set_modal("win")
	_sfx("win")
	_narrate("complete")
	_save_progress()


func _notify(message: String, duration := 4.0) -> void:
	toast = message
	toast_time = duration
	toast_duration = duration


func _hint() -> void:
	hint_level += 1
	var message := ""
	if not mirror_found:
		message = "看看左侧案上的手记，再找找桌边那面小铜镜。"
	elif not mirror_placed:
		message = "点击行囊中的铜镜，再点击画面中央的空镜台。"
	elif not revealed:
		message = "光应落在右上方圆盘。" + ("从最初角度向右转四次。" if hint_level > 2 else "试着慢慢向右转动镜片。")
	elif not cabinet_open:
		message = "按从左到右的顺序：山 → 月 → 水，调整旧匣上的三枚纹样。"
	else:
		message = "旧匣已经打开。点击匣中的青瓷残片，完成本章。"
	_notify(message, 6)


func _save_progress() -> void:
	if transient_save:
		return
	var data := {"version": 1, "mirror_found": mirror_found, "mirror_placed": mirror_placed, "note_read": note_read, "revealed": revealed, "cabinet_open": cabinet_open, "completed": completed, "angle_step": angle_step, "dials": dials}
	var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	if file:
		file.store_string(JSON.stringify(data))


func _load_progress() -> void:
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var data = JSON.parse_string(FileAccess.get_file_as_string(SAVE_PATH))
	if not data is Dictionary or int(data.get("version", 0)) != 1:
		return
	mirror_found = bool(data.get("mirror_found", false))
	mirror_placed = mirror_found and bool(data.get("mirror_placed", false))
	note_read = bool(data.get("note_read", false))
	revealed = mirror_placed and bool(data.get("revealed", false))
	cabinet_open = revealed and bool(data.get("cabinet_open", false))
	completed = cabinet_open and bool(data.get("completed", false))
	angle_step = clampi(int(data.get("angle_step", 0)), 0, 10)
	var stored = data.get("dials", [3, 4, 5])
	if stored is Array and stored.size() == 3:
		for i in range(3):
			dials[i] = posmod(int(stored[i]), 6)


func _reset_progress() -> void:
	mirror_found = false
	mirror_placed = false
	note_read = false
	revealed = false
	cabinet_open = false
	completed = false
	angle_step = 0
	dials = [3, 4, 5]
	selected_item = false
	modal = ""
	hint_level = 0
	play_time = 0.0
	lid_t = 0.0
	angle_vis = 0.0
	toast_time = 0
	speech.stop()
	_save_progress()


func _expect(condition: bool, label: String) -> void:
	if not condition:
		push_error("FAIL: " + label)
		get_tree().quit(1)
		assert(condition, label)
	print("PASS: " + label)


func _run_self_test() -> void:
	_reset_progress()
	_expect(not mirror_found and not revealed and not completed, "new game has clean progress")
	_act("stand")
	_expect(not mirror_placed, "empty inventory cannot place mirror")
	dials = ANSWER.duplicate()
	_check_seal()
	_expect(not cabinet_open, "seal needs discovered evidence even with correct symbols")
	_act("note")
	_act("close")
	_expect(note_read, "note becomes persistent clue")
	_act("pickup")
	_expect(mirror_found and selected_item, "collect mirror and select inventory")
	_act("stand")
	_expect(mirror_placed and not selected_item, "mirror used on correct stand")
	_turn_mirror(1)
	_expect(not revealed, "incorrect beam angle does not reveal clue")
	for i in range(3):
		_turn_mirror(1)
	_expect(revealed and angle_step == 4, "reflected ray intersects target and reveals symbols")
	dials = [2, 1, 0]
	_check_seal()
	_expect(not cabinet_open, "wrong symbol order rejected")
	dials = ANSWER.duplicate()
	_check_seal()
	_expect(cabinet_open and not completed, "correct order opens casket but requires collecting artifact")
	_act("chest")
	_expect(completed and modal == "win", "artifact collection completes chapter")
	_reset_progress()
	_expect(not revealed and not mirror_placed and dials == [3, 4, 5], "restart resets all puzzles")
	print("SELF_TEST_OK: 12 gameplay checks")
	get_tree().quit(0)


func _capture() -> void:
	_reset_progress()
	if capture_stage != "title":
		screen = "game"
	if capture_stage in ["light", "seal", "win"]:
		mirror_found = true
		mirror_placed = true
		note_read = true
		angle_step = 4
		revealed = true
	if capture_stage == "note":
		note_read = true
		modal = "note"
	if capture_stage == "mirror":
		mirror_found = true
		modal = "mirror"
	if capture_stage == "seal":
		modal = "seal"
	if capture_stage == "win":
		cabinet_open = true
		completed = true
		modal = "win"
	lid_t = 1.0 if cabinet_open else 0.0
	await get_tree().create_timer(2.0).timeout
	await RenderingServer.frame_post_draw
	var result := get_viewport().get_texture().get_image().save_png(capture_path)
	print("CAPTURE_RESULT: ", result)
	get_tree().quit(0 if result == OK else 1)


func _click_at(p: Vector2) -> void:
	var press := InputEventMouseButton.new()
	press.position = p
	press.global_position = p
	press.button_index = MOUSE_BUTTON_LEFT
	press.pressed = true
	get_viewport().push_input(press, true)
	var release := press.duplicate() as InputEventMouseButton
	release.pressed = false
	get_viewport().push_input(release, true)
	await get_tree().process_frame
	await RenderingServer.frame_post_draw
